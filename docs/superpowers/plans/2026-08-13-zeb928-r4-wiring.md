**Goal:** Wire the pure ZEB-914 topology engine into the live dial path so per-community
degree is bounded in router mode, via a node_id-keyed admission choke point.

**Architecture:** Three parts — a resolver-side reverse bridge `node_id → device_key(s)`, an
event-loop controller that recomputes the admitted *device-key* set on membership deltas, and a
shared `AdmissionOracle` read at the supervisor's two dial-arming sites (`kick` + `do_sweep`).
See `docs/superpowers/specs/2026-08-13-zeb928-r4-wiring-design.md`.

**Tech Stack:** Rust, `harmony-app` crate under `src-tauri/`. Engine at
`src/community_topology.rs`.

// src-tauri/src/admission_oracle.rs  (top of file)
//! Node-id admission classifier for R4 bounded-degree dialing (ZEB-928).
//!
//! Read on the hot path at the supervisor's two dial-arming sites. Peer mode (or an
//! unwired oracle) admits everything — behavior identical to pre-R4. Router mode admits a
//! node_id iff a device key bound to it is in the admitted set; unknown node_ids fail open.
//! See `docs/superpowers/specs/2026-08-13-zeb928-r4-wiring-design.md`.

use std::collections::{BTreeSet, HashMap};
use std::sync::RwLock;

pub struct AdmissionOracle {
    enabled: bool,
    admitted: RwLock<BTreeSet<[u8; 32]>>,                    // device keys (controller writes)
    node_to_devices: RwLock<HashMap<[u8; 32], BTreeSet<[u8; 32]>>>, // reverse bridge (resolver writes)
}

impl AdmissionOracle {
    /// `enabled` = router mode. Disabled oracle admits everything.
    pub fn new(enabled: bool) -> Self {
        Self {
            enabled,
            admitted: RwLock::new(BTreeSet::new()),
            node_to_devices: RwLock::new(HashMap::new()),
        }
    }

    pub fn enabled(&self) -> bool {
        self.enabled
    }

    /// Hot path. True = keep/allow this dial target.
    pub fn admit(&self, node_id: &[u8; 32]) -> bool {
        if !self.enabled {
            return true;
        }
        let map = self.node_to_devices.read().expect("node_to_devices poisoned");
        let devices = match map.get(node_id) {
            Some(d) => d,
            None => return true, // fail-open on unknown identity
        };
        let admitted = self.admitted.read().expect("admitted poisoned");
        devices.iter().any(|d| admitted.contains(d))
    }

    /// Controller: replace the admitted device-key set (membership delta).
    pub fn publish_admitted(&self, keys: BTreeSet<[u8; 32]>) {
        *self.admitted.write().expect("admitted poisoned") = keys;
    }

    /// Resolver: record that `node_id` is asserted by `device_key`.
    pub fn bind(&self, node_id: [u8; 32], device_key: [u8; 32]) {
        self.node_to_devices
            .write()
            .expect("node_to_devices poisoned")
            .entry(node_id)
            .or_default()
            .insert(device_key);
    }

    /// Resolver `remove_owner`: forget these node_ids entirely.
    pub fn unbind_node_ids(&self, node_ids: &[[u8; 32]]) {
        let mut map = self.node_to_devices.write().expect("node_to_devices poisoned");
        for n in node_ids {
            map.remove(n);
        }
    }
}
```

// src-tauri/src/admission_oracle.rs  (above the tests module)
use crate::community_topology::community_neighbors;

/// Union of chosen ring neighbors across the joined communities. Each entry is
/// `(active enrolled device keys including self, community salt bytes)`.
///
/// Policy: when `self_vk` is not on a community's ring (join window before local
/// enrollment materializes), that community contributes its full active device set —
/// prevents islanding. Below `FULL_MESH_THRESHOLD`, `community_neighbors` already returns
/// all-but-self. `self_vk` is never included in the result.
pub fn compute_admitted(
    communities: &[(BTreeSet<[u8; 32]>, Vec<u8>)],
    self_vk: &[u8; 32],
) -> BTreeSet<[u8; 32]> {
    let mut out = BTreeSet::new();
    for (devices, salt) in communities {
        if devices.contains(self_vk) {
            out.extend(community_neighbors(devices, self_vk, salt));
        } else {
            // self not on ring: admit the whole active set for this community.
            out.extend(devices.iter().copied());
        }
    }
    out.remove(self_vk);
    out
}
```

    let registry = std::sync::Arc::clone(&registry);
    let self_vk = community_signing_key_arc.verifying_key().to_bytes();
    tokio::spawn(async move {
        if !oracle.enabled() { return; } // peer mode: no filtering, no polling
        let mut seen_versions: std::collections::HashMap<SpaceId, u64> = Default::default();
        let mut tick = tokio::time::interval(std::time::Duration::from_secs(2));
        loop {
            tick.tick().await;
            let ids = registry.spawned_community_ids().await;
            let mut changed = false;
            let mut communities: Vec<(std::collections::BTreeSet<[u8;32]>, Vec<u8>)> = Vec::new();
            for id in &ids {
                let Some(engine) = registry.engine_arc(id).await else { continue };
                let (ver, devices) = {
                    let st = engine.state().lock().await;
                    let ver = st.materialized_version();
                    let mat = st.materialized(engine.admin_addr());
                    let devices: std::collections::BTreeSet<[u8;32]> =
                        crate::community_gateway_dial_driver::enrolled_keys_from_members(&mat.members)
                            .into_iter().collect();
                    (ver, devices)
                };
                if seen_versions.get(id) != Some(&ver) { changed = true; seen_versions.insert(*id, ver); }
                communities.push((devices, id.0.to_vec()));
            }
            // also recompute if the joined-set shrank
            if ids.len() != seen_versions.len() { changed = true; seen_versions.retain(|k,_| ids.contains(k)); }
            if changed {
                let admitted = crate::admission_oracle::compute_admitted(&communities, &self_vk);
                oracle.publish_admitted(admitted);
                supervisor_handle.kick_sweep(); // re-arm; filtered sweep dials the admitted set
            }
        }
    });
}
